11.1 Geração de Modelos Locais - LLM com LmStudio

11.1.1 1 - Crie uma rede neural transformers LLM GPT2

python modelo-transformer-gpt2.py
import os
from transformers import AutoConfig, AutoModelForCausalLM, AutoTokenizer

# 1. Definir uma configuração ultraleve (para fins puramente didáticos e rápidos)
# Usaremos a arquitetura GPT2 como base por ser simples e amplamente suportada
config = AutoConfig.from_pretrained("gpt2")
config.n_embd = 128  # Dimensão do embedding bem pequena
config.n_layer = 2  # Apenas 2 camadas de Transformer
config.n_head = 2  # 2 cabeças de atenção
config.n_positions = 1024
config.vocab_size = 50257

print("Criando o modelo Transformer com a arquitetura customizada...")
model = AutoModelForCausalLM.from_config(config)
tokenizer = AutoTokenizer.from_pretrained("gpt2")

# 2. Salvar no formato padrão do Hugging Face em uma pasta local
output_dir = "./meu_mini_transformer"
os.makedirs(output_dir, exist_ok=True)

model.save_pretrained(output_dir)
tokenizer.save_pretrained(output_dir)

print(f"Modelo salvo com sucesso em '{output_dir}'!")
